local function remote(name)
    return NET and NET:FindFirstChild(name)
end

local function fire(name, ...)
    local r = remote(name)
    if not r then return false, "missing " .. name end
    local ok, err = pcall(function(...)
        if r:IsA("RemoteEvent") then
            r:FireServer(...)
        elseif r:IsA("RemoteFunction") then
            return r:InvokeServer(...)
        end
    end, ...)
    return ok, err
end

local function invoke(name, ...)
    local r = remote(name)
    if not r or not r:IsA("RemoteFunction") then return nil, "missing RF " .. name end
    local ok, res = pcall(function(...)
        return r:InvokeServer(...)
    end, ...)
    if ok then return res end
    return nil, res
end

local function notify(title, content, dur)
    Library:Notify({ Title = title, Content = tostring(content), Duration = dur or 4 })
end

local lastSnapshot = nil

local function refreshSnapshot()
    local snap, err = invoke("GetMatchSnapshot")
    if snap then
        lastSnapshot = snap
        return snap
    end
    return nil, err
end

local function snapshotSummary()
    local s = lastSnapshot or select(1, refreshSnapshot())
    if type(s) ~= "table" then
        return "No snapshot (try again in-match)"
    end
    -- best-effort fields; structure may vary
    local lines = {}
    local function add(k, v)
        if v ~= nil then table.insert(lines, tostring(k) .. ": " .. tostring(v)) end
    end
    add("phase", s.phase or s.Phase or s.state or s.State)
    add("turn", s.turn or s.Turn or s.currentTurn)
    add("active", s.activeUserId or s.ActiveUserId or s.currentPlayer)
    if type(s.players) == "table" then
        add("players", #s.players)
    elseif type(s.Players) == "table" then
        add("players", #s.Players)
    end
    -- cash for local if present
    local me = s.players or s.Players or s.playerStates
    if type(me) == "table" then
        for _, p in pairs(me) do
            if type(p) == "table" then
                local uid = p.userId or p.UserId or p.id
                if uid == LocalPlayer.UserId then
                    add("cash", p.cash or p.Cash or p.money or p.Money)
                    add("job", p.job or p.Job or p.career)
                    add("tile", p.tileId or p.TileId or p.position)
                    break
                end
            end
        end
    end
    if #lines == 0 then
        -- dump top-level keys for debugging
        local keys = {}
        for k in pairs(s) do table.insert(keys, tostring(k)) end
        table.sort(keys)
        return "Keys: " .. table.concat(keys, ", ")
    end
    return table.concat(lines, "\n")
end

local ESPFolder = Instance.new("Folder")
ESPFolder.Name = "GoL_EvolESP"
ESPFolder.Parent = game:GetService("CoreGui")

local tileMarks = {}
local pawnMarks = {}

local function clearFolder(map)
    for _, v in pairs(map) do
        pcall(function() v:Destroy() end)
    end
    table.clear(map)
end

local function markPart(part, color, text, store)
    if not part or not part:IsA("BasePart") then return end
    local hl = Instance.new("Highlight")
    hl.Adornee = part.Parent:IsA("Model") and part.Parent or part
    hl.FillColor = color
    hl.OutlineColor = Color3.new(1, 1, 1)
    hl.FillTransparency = 0.6
    hl.OutlineTransparency = 0.15
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = ESPFolder

    local bb = Instance.new("BillboardGui")
    bb.Size = UDim2.fromOffset(140, 28)
    bb.StudsOffset = Vector3.new(0, 2.5, 0)
    bb.AlwaysOnTop = true
    bb.Adornee = part
    bb.Parent = ESPFolder
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.fromScale(1, 1)
    lbl.BackgroundTransparency = 1
    lbl.Text = text
    lbl.TextColor3 = color
    lbl.TextStrokeTransparency = 0.4
    lbl.Font = Enum.Font.GothamBold
    lbl.TextSize = 12
    lbl.Parent = bb

    store[part] = hl
    store[part:GetDebugId() .. "_bb"] = bb
end

local function scanTiles()
    clearFolder(tileMarks)
    if not CFG.TileESP then return end

    local board = Workspace:FindFirstChild("GameOfLifeBoard")
    if not board then return end
    local tiles = board:FindFirstChild("Tiles")
    if not tiles then return end

    local function walk(folder)
        for _, child in ipairs(folder:GetChildren()) do
            if child:IsA("Model") then
                local family = child:GetAttribute("TileFamily")
                local tileId = child:GetAttribute("TileId") or child.Name
                local resolver = child:GetAttribute("ResolverId")
                local isAction = family == "Action" or (child:GetAttribute("ActionTileIconInstalled") == true)
                local tags = {}
                pcall(function()
                    for _, t in ipairs(child:GetTags()) do table.insert(tags, t) end
                end)

                local show = true
                if CFG.ActionTileESP and not isAction and family ~= "Action" then
                    -- still show other interesting families
                    if family == nil and not table.find(tags, "ActionTile") then
                        show = false
                    end
                end

                if show then
                    local part = child:FindFirstChildWhichIsA("BasePart", true)
                    if part then
                        local color = isAction and Color3.fromRGB(255, 140, 40)
                            or (family == "Start" and Color3.fromRGB(80, 220, 120))
                            or Color3.fromRGB(100, 180, 255)
                        local label = CFG.ShowTileId and tostring(tileId) or (family or "Tile")
                        if resolver then label = label .. " · " .. tostring(resolver) end
                        markPart(part, color, label, tileMarks)
                    end
                end
            elseif child:IsA("Folder") then
                walk(child)
            end
        end
    end
    walk(tiles)
end

local function scanPawns()
    clearFolder(pawnMarks)
    if not CFG.PawnESP then return end

    -- pawns often live under board or as player models near tiles
    local board = Workspace:FindFirstChild("GameOfLifeBoard")
    local roots = { board, Workspace }

    for _, root in ipairs(roots) do
        if not root then continue end
        for _, inst in ipairs(root:GetDescendants()) do
            if inst:IsA("Model") and (inst.Name:lower():find("pawn") or inst:GetAttribute("UserId") or inst:GetAttribute("userId")) then
                local part = inst.PrimaryPart or inst:FindFirstChildWhichIsA("BasePart")
                if part then
                    local uid = inst:GetAttribute("UserId") or inst:GetAttribute("userId")
                    local name = inst.Name
                    if uid then
                        local plr = Players:GetPlayerByUserId(tonumber(uid) or 0)
                        if plr then name = plr.DisplayName end
                    end
                    local col = (uid == LocalPlayer.UserId) and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(255, 90, 90)
                    markPart(part, col, name, pawnMarks)
                end
            end
        end
    end

    -- also other players' characters if present
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character then
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local col = plr == LocalPlayer and Color3.fromRGB(80, 255, 120) or Color3.fromRGB(255, 90, 90)
                markPart(hrp, col, plr.DisplayName, pawnMarks)
            end
        end
    end
end

task.spawn(function()
    while true do
        task.wait(2)
        pcall(scanTiles)
        pcall(scanPawns)
    end
end)

local oldLit = {}
local function setFullbright(on)
    if on then
        if not oldLit.ok then
            oldLit = {
                ok = true,
                Brightness = Lighting.Brightness,
                ClockTime = Lighting.ClockTime,
                FogEnd = Lighting.FogEnd,
                GlobalShadows = Lighting.GlobalShadows,
                Ambient = Lighting.Ambient,
            }
        end
        Lighting.Brightness = 2
        Lighting.ClockTime = 14
        Lighting.FogEnd = 1e6
        Lighting.GlobalShadows = false
        Lighting.Ambient = Color3.new(1, 1, 1)
    elseif oldLit.ok then
        Lighting.Brightness = oldLit.Brightness
        Lighting.ClockTime = oldLit.ClockTime
        Lighting.FogEnd = oldLit.FogEnd
        Lighting.GlobalShadows = oldLit.GlobalShadows
        Lighting.Ambient = oldLit.Ambient
        oldLit = {}
    end
end
