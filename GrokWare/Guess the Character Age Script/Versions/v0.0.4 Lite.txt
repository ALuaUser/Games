----------------------------------------------------------------
-- Load CharacterConfig
----------------------------------------------------------------
local CharacterConfig

local function tryRequire(parts)
    local n = RS
    for _, p in ipairs(parts) do
        n = n and n:FindFirstChild(p)
    end
    if n and n:IsA("ModuleScript") then
        local ok, mod = pcall(require, n)
        if ok and type(mod) == "table" then
            return mod
        end
    end
end

for _, path in ipairs({
    { "CharacterConfig" },
    { "Modules", "CharacterConfig" },
    { "Shared", "CharacterConfig" },
    { "Configs", "CharacterConfig" },
    { "Config", "CharacterConfig" },
    { "Data", "CharacterConfig" },
}) do
    CharacterConfig = tryRequire(path)
    if CharacterConfig then
        break
    end
end

if not CharacterConfig then
    for _, d in ipairs(RS:GetDescendants()) do
        if d:IsA("ModuleScript") and d.Name:lower():find("character") then
            local ok, mod = pcall(require, d)
            if ok and type(mod) == "table" then
                local k = next(mod)
                if k and type(mod[k]) == "table" and mod[k].Age ~= nil then
                    CharacterConfig = mod
                    break
                end
            end
        end
    end
end

local function lookup(name)
    if not CharacterConfig or not name then
        return
    end
    name = tostring(name):gsub("^%s+", ""):gsub("%s+$", "")
    if CharacterConfig[name] then
        return CharacterConfig[name], name
    end
    local lower = name:lower()
    for n, data in pairs(CharacterConfig) do
        if n:lower() == lower then
            return data, n
        end
    end
    for n, data in pairs(CharacterConfig) do
        if n:lower():find(lower, 1, true) or lower:find(n:lower(), 1, true) then
            return data, n
        end
    end
end

----------------------------------------------------------------
-- Heading text
----------------------------------------------------------------
local function headingText()
    local gui = PG:FindFirstChild("GameGui")
    local h = gui and gui:FindFirstChild("Heading")
    if not h then
        return
    end
    if h:IsA("TextLabel") or h:IsA("TextButton") or h:IsA("TextBox") then
        return h.Text
    end
    local lab = h:FindFirstChildWhichIsA("TextLabel", true)
        or h:FindFirstChildWhichIsA("TextButton", true)
    return lab and lab.Text
end

----------------------------------------------------------------
-- Drag
----------------------------------------------------------------
do
    local dragging, start, startPos
    card.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            start = input.Position
            startPos = card.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - start
            card.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

----------------------------------------------------------------
-- Update logic
----------------------------------------------------------------
local lastName

local function setAnswer(realName, data)
    if not data then
        nameLbl.Text = "Waiting for character…"
        workLbl.Text = CharacterConfig and "Config loaded" or "CharacterConfig missing"
        ageLbl.Text = "Age  —"
        ageLbl.TextColor3 = Color3.fromRGB(160, 170, 185)
        accent.BackgroundColor3 = Color3.fromRGB(90, 200, 255)
        return
    end

    nameLbl.Text = realName
    workLbl.Text = data.Work or ""
    ageLbl.Text = "Age  " .. tostring(data.Age)
    ageLbl.TextColor3 = Color3.fromRGB(100, 255, 180)
    accent.BackgroundColor3 = Color3.fromRGB(80, 255, 170)

    if realName ~= lastName then
        lastName = realName
        ageLbl.TextTransparency = 1
        TweenService:Create(ageLbl, TweenInfo.new(0.25), { TextTransparency = 0 }):Play()
    end
end

local function refresh()
    local text = headingText()
    if not text or #text < 2 then
        setAnswer(nil, nil)
        return
    end
    local data, real = lookup(text)
    if data then
        setAnswer(real, data)
    else
        nameLbl.Text = text
        workLbl.Text = "Not in CharacterConfig"
        ageLbl.Text = "Age  ?"
        ageLbl.TextColor3 = Color3.fromRGB(255, 180, 90)
        accent.BackgroundColor3 = Color3.fromRGB(255, 170, 70)
    end
end

-- poll + hook Heading when it appears
task.spawn(function()
    while gui.Parent do
        refresh()
        local g = PG:FindFirstChild("GameGui")
        local h = g and g:FindFirstChild("Heading")
        if h and not h:GetAttribute("LiteHooked") then
            h:SetAttribute("LiteHooked", true)
            local function hook(obj)
                if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                    obj:GetPropertyChangedSignal("Text"):Connect(refresh)
                end
            end
            hook(h)
            for _, d in ipairs(h:GetDescendants()) do
                hook(d)
            end
            h.DescendantAdded:Connect(hook)
        end
        task.wait(0.3)
    end
end)
