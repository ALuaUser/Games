local TOGGLE_KEY = Enum.KeyCode.RightShift

----------------------------------------------------------------
-- Load CharacterConfig
----------------------------------------------------------------
local CharacterConfig

local function tryRequire(parts)
    local n = RS
    for _, p in ipairs(parts) do
        n = n and n:FindFirstChild(p)
    end
    if n and n:IsA("ModuleScript") then
        local ok, mod = pcall(require, n)
        if ok and type(mod) == "table" then
            return mod
        end
    end
end

for _, path in ipairs({
    { "CharacterConfig" },
    { "Modules", "CharacterConfig" },
    { "Shared", "CharacterConfig" },
    { "Configs", "CharacterConfig" },
    { "Config", "CharacterConfig" },
    { "Data", "CharacterConfig" },
}) do
    CharacterConfig = tryRequire(path)
    if CharacterConfig then
        break
    end
end

if not CharacterConfig then
    for _, d in ipairs(RS:GetDescendants()) do
        if d:IsA("ModuleScript") and d.Name:lower():find("character") then
            local ok, mod = pcall(require, d)
            if ok and type(mod) == "table" then
                local k = next(mod)
                if k and type(mod[k]) == "table" and mod[k].Age ~= nil then
                    CharacterConfig = mod
                    break
                end
            end
        end
    end
end

local function lookup(name)
    if not CharacterConfig or not name then
        return
    end
    name = tostring(name):gsub("^%s+", ""):gsub("%s+$", "")
    if CharacterConfig[name] then
        return CharacterConfig[name], name
    end
    local lower = name:lower()
    for n, data in pairs(CharacterConfig) do
        if n:lower() == lower then
            return data, n
        end
    end
    for n, data in pairs(CharacterConfig) do
        if n:lower():find(lower, 1, true) or lower:find(n:lower(), 1, true) then
            return data, n
        end
    end
end

----------------------------------------------------------------
-- Heading text
----------------------------------------------------------------
local function headingText()
    local gui = PG:FindFirstChild("GameGui")
    local h = gui and gui:FindFirstChild("Heading")
    if not h then
        return
    end
    if h:IsA("TextLabel") or h:IsA("TextButton") or h:IsA("TextBox") then
        return h.Text
    end
    local lab = h:FindFirstChildWhichIsA("TextLabel", true)
        or h:FindFirstChildWhichIsA("TextButton", true)
    return lab and lab.Text
end

local C = {
    bg = Color3.fromRGB(20, 21, 24),
    line = Color3.fromRGB(255, 255, 255),
    text = Color3.fromRGB(236, 237, 240),
    sub = Color3.fromRGB(135, 139, 148),
    accent = Color3.fromRGB(120, 200, 150),
    warn = Color3.fromRGB(224, 175, 104),
}

local FULL_H, MINI_H, WIDTH = 72, 36, 264

local function tween(obj, props, t, style, dir)
    local tw = TweenService:Create(
        obj,
        TweenInfo.new(t or 0.2, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    )
    tw:Play()
    return tw
end

local function make(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do
        o[k] = v
    end
    o.Parent = parent
    return o
end

-- header buttons (show on hover)
local function headerButton(text, xOffset)
    local b = make("TextButton", {
        BackgroundTransparency = 1,
        Position = UDim2.new(1, xOffset, 0, 3),
        Size = UDim2.fromOffset(18, 18),
        Font = Enum.Font.BuilderSansBold,
        TextSize = 16,
        Text = text,
        TextColor3 = C.sub,
        TextTransparency = 1,
        AutoButtonColor = false,
        ZIndex = 5,
    }, card)
    b.MouseEnter:Connect(function()
        tween(b, { TextColor3 = C.text }, 0.12)
    end)
    b.MouseLeave:Connect(function()
        tween(b, { TextColor3 = C.sub }, 0.12)
    end)
    return b
end

local closeBtn = headerButton("×", -22)
local miniBtn = headerButton("–", -42)

local hovering = false
card.MouseEnter:Connect(function()
    hovering = true
    tween(closeBtn, { TextTransparency = 0 }, 0.15)
    tween(miniBtn, { TextTransparency = 0 }, 0.15)
end)
card.MouseLeave:Connect(function()
    hovering = false
    tween(closeBtn, { TextTransparency = 1 }, 0.15)
    tween(miniBtn, { TextTransparency = 1 }, 0.15)
end)

----------------------------------------------------------------
-- Show / hide animation
----------------------------------------------------------------
local shown = false
local restPos = card.Position -- where the card "lives" (updated by drag)

local function setShown(on, onDone)
    shown = on
    if on then
        gui.Enabled = true
        card.Position = restPos + UDim2.fromOffset(0, -8)
    end

    local t = on and 0 or 1
    local dur = on and 0.3 or 0.22
    local style = on and Enum.EasingStyle.Quad or Enum.EasingStyle.Quint

    tween(card, {
        BackgroundTransparency = on and 0.04 or 1,
        Position = on and restPos or (restPos + UDim2.fromOffset(0, 8)),
    }, dur, style)
    tween(stroke, { Transparency = on and 0.9 or 1 }, dur)
    tween(divider, { BackgroundTransparency = on and 0.9 or 1 }, dur)
    tween(nameLbl, { TextTransparency = t }, dur)
    tween(workLbl, { TextTransparency = t }, dur)
    tween(ageLbl, { TextTransparency = t }, dur)
    if not on then
        tween(closeBtn, { TextTransparency = 1 }, dur)
        tween(miniBtn, { TextTransparency = 1 }, dur)
    end

    task.delay(dur, function()
        if not on and not shown then
            gui.Enabled = false
        end
        if onDone then
            onDone()
        end
    end)
end

closeBtn.MouseButton1Click:Connect(function()
    setShown(false, function()
        gui:Destroy()
    end)
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if processed or input.KeyCode ~= TOGGLE_KEY then
        return
    end
    setShown(not shown)
end)

----------------------------------------------------------------
-- Drag
----------------------------------------------------------------
do
    local dragging, start, startPos
    card.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            start = input.Position
            startPos = card.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - start
            local cam = workspace.CurrentCamera
            local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
            restPos = UDim2.fromOffset(
                math.clamp(startPos.X.Offset + d.X, 0, vp.X - card.AbsoluteSize.X),
                math.clamp(startPos.Y.Offset + d.Y, 0, vp.Y - card.AbsoluteSize.Y)
            )
            card.Position = restPos
        end
    end)
end

----------------------------------------------------------------
-- Update logic
----------------------------------------------------------------
local lastName

local function setAnswer(realName, data)
    if not data then
        nameLbl.Text = "Waiting…"
        workLbl.Text = CharacterConfig and "" or "CharacterConfig missing"
        ageLbl.Text = "–"
        tween(ageLbl, { TextColor3 = C.sub }, 0.2)
        return
    end

    nameLbl.Text = realName
    workLbl.Text = data.Work or ""
    ageLbl.Text = tostring(data.Age)
    tween(ageLbl, { TextColor3 = C.accent }, 0.2)

    if realName ~= lastName then
        lastName = realName
        if shown then
            ageLbl.TextTransparency = 1
            tween(ageLbl, { TextTransparency = 0 }, 0.25)
        end
    end
end

local function refresh()
    local text = headingText()
    if not text or #text < 2 then
        setAnswer(nil, nil)
        return
    end
    local data, real = lookup(text)
    if data then
        setAnswer(real, data)
    else
        nameLbl.Text = text
        workLbl.Text = "Not in CharacterConfig"
        ageLbl.Text = "?"
        tween(ageLbl, { TextColor3 = C.warn }, 0.2)
    end
end

-- poll + hook Heading when it appears
task.spawn(function()
    while gui.Parent do
        refresh()
        local g = PG:FindFirstChild("GameGui")
        local h = g and g:FindFirstChild("Heading")
        if h and not h:GetAttribute("LiteHooked") then
            h:SetAttribute("LiteHooked", true)
            local function hook(obj)
                if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                    obj:GetPropertyChangedSignal("Text"):Connect(refresh)
                end
            end
            hook(h)
            for _, d in ipairs(h:GetDescendants()) do
                hook(d)
            end
            h.DescendantAdded:Connect(hook)
        end
        task.wait(0.3)
    end
end)

setShown(true)
