-- Load game modules
----------------------------------------------------------------
local function tryRequire(pathParts)
    local node = RS
    for _, name in ipairs(pathParts) do
        node = node and node:FindFirstChild(name)
    end
    if node and node:IsA("ModuleScript") then
        local ok, mod = pcall(require, node)
        if ok and type(mod) == "table" then
            return mod
        end
    end
    return nil
end

local function loadConfigs()
    local charPaths = {
        { "CharacterConfig" },
        { "Modules", "CharacterConfig" },
        { "Shared", "CharacterConfig" },
        { "Configs", "CharacterConfig" },
        { "Config", "CharacterConfig" },
        { "Data", "CharacterConfig" },
    }
    local skinPaths = {
        { "CharSkins" },
        { "Modules", "CharSkins" },
        { "Shared", "CharSkins" },
        { "Configs", "CharSkins" },
        { "Config", "CharSkins" },
        { "Data", "CharSkins" },
    }

    for _, p in ipairs(charPaths) do
        CharacterConfig = tryRequire(p)
        if CharacterConfig then
            break
        end
    end
    for _, p in ipairs(skinPaths) do
        CharSkins = tryRequire(p)
        if CharSkins then
            break
        end
    end

    if not CharacterConfig then
        for _, desc in ipairs(RS:GetDescendants()) do
            if desc:IsA("ModuleScript") and string.lower(desc.Name):find("character") then
                local ok, mod = pcall(require, desc)
                if ok and type(mod) == "table" then
                    local sample = next(mod)
                    if sample and type(mod[sample]) == "table" and mod[sample].Age ~= nil then
                        CharacterConfig = mod
                        break
                    end
                end
            end
        end
    end
end

loadConfigs()

    end
    table.sort(nameList)
end

----------------------------------------------------------------
-- Helpers
----------------------------------------------------------------
local function notify(title, content, dur)
    Library:Notify({
        Title = title or "Age Guess",
        Content = content or "",
        Duration = dur or 4,
    })
end

local function getChar(name)
    if not CharacterConfig or not name then
        return nil
    end
    if CharacterConfig[name] then
        return CharacterConfig[name], name
    end
    local lower = string.lower(name)
    for n, data in pairs(CharacterConfig) do
        if string.lower(n) == lower then
            return data, n
        end
    end
    for n, data in pairs(CharacterConfig) do
        if string.find(string.lower(n), lower, 1, true) then
            return data, n
        end
    end
    return nil
end

local function fireEvent(name, ...)
    if not Events then
        return false
    end
    local ev = Events:FindFirstChild(name)
    if not (ev and ev:IsA("RemoteEvent")) then
        return false
    end
    local args = table.pack(...)
    local ok = pcall(function()
        ev:FireServer(table.unpack(args, 1, args.n))
    end)
    return ok
end

----------------------------------------------------------------
-- Detect current character from UI
----------------------------------------------------------------
local currentName = nil
local currentAge = nil

local function scrapeCharacterName()
    if not CharacterConfig then
        return nil
    end
    for _, gui in ipairs(PG:GetDescendants()) do
        if gui:IsA("TextLabel") or gui:IsA("TextButton") or gui:IsA("TextBox") then
            local text = gui.Text
            if type(text) == "string" and #text > 2 and #text < 64 then
                local data, real = getChar(text)
                if data then
                    return real, data
                end
            end
        end
    end
    return nil
end

local function findGuessBox()
    for _, gui in ipairs(PG:GetDescendants()) do
        if gui:IsA("TextBox") then
            local n = string.lower(gui.Name .. tostring(gui.PlaceholderText or ""))
            if n:find("age") or n:find("guess") or n:find("answer") or n:find("input") then
                return gui
            end
        end
    end
    for _, gui in ipairs(PG:GetDescendants()) do
        if gui:IsA("TextBox") and gui.Visible then
            return gui
        end
    end
    return nil
end

local function findSubmitButton()
    for _, gui in ipairs(PG:GetDescendants()) do
        if gui:IsA("TextButton") or gui:IsA("ImageButton") then
            local t = string.lower(tostring(gui.Text or "") .. gui.Name)
            if t:find("submit") or t:find("guess") or t:find("confirm") or t:find("send") or t:find("enter") then
                return gui
            end
        end
    end
    return nil
end

local function applyGuess(age, submit)
    local box = findGuessBox()
    if box then
        box.Text = tostring(age)
    end
    if submit then
        task.wait(CFG.SubmitDelay)
        local btn = findSubmitButton()
        if btn then
            pcall(function()
                firesignal(btn.MouseButton1Click)
            end)
            pcall(function()
                btn:Activate()
            end)
        end
        fireEvent("RemoteEvent", age)
        fireEvent("Purchase", age)
        fireEvent("Purchase2", age)
    end
end

local function updateHUD(name, data)
    currentName = name
    currentAge = data and data.Age or nil
    if name and data then
        hudTitle.Text = data.Work and (name .. "  ·  " .. data.Work) or name
        hudAge.Text = "Age: " .. tostring(data.Age)
    else
        hudTitle.Text = "Character: —"
        hudAge.Text = "Age: ?"
    end
end

----------------------------------------------------------------
-- Detect + auto guess loop
----------------------------------------------------------------
task.spawn(function()
    local lastApplied = nil
    while task.wait(0.35) do
        hud.Visible = CFG.ShowHUD
        local name, data = scrapeCharacterName()
        if name and data then
            updateHUD(name, data)
            if CFG.AutoGuess and lastApplied ~= name then
                lastApplied = name
                applyGuess(data.Age, CFG.AutoSubmit)
                notify("Auto Guess", name .. " → " .. tostring(data.Age), 3)
            end
        end
    end
end)
