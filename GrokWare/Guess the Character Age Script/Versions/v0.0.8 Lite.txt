local FILE = "AgeHUD_Lite_settings.json"
local DEFAULTS = {
    x = 18,
    y = 120,
    opacity = 0.96,
    scale = 1,
    theme = "dark",
    key = "RightShift",
    history = false,
    mini = false,
}

local function loadSettings()
    local raw
    if type(isfile) == "function" and type(readfile) == "function" then
        local ok, res = pcall(function()
            if isfile(FILE) then
                return readfile(FILE)
            end
        end)
        if ok then
            raw = res
        end
    end
    if not raw then
        raw = LP:GetAttribute("AgeHUDLiteSettings")
    end

    local out = table.clone(DEFAULTS)
    if type(raw) == "string" then
        local ok, data = pcall(HttpService.JSONDecode, HttpService, raw)
        if ok and type(data) == "table" then
            for k, def in pairs(DEFAULTS) do
                if type(data[k]) == type(def) then
                    out[k] = data[k]
                end
            end
        end
    end

    out.opacity = math.clamp(out.opacity, 0.5, 1)
    out.scale = math.clamp(out.scale, 0.8, 1.3)
    if out.theme ~= "dark" and out.theme ~= "light" then
        out.theme = "dark"
    end
    if not pcall(function()
        return Enum.KeyCode[out.key]
    end) then
        out.key = DEFAULTS.key
    end
    return out
end

local S = loadSettings()

local saveToken = 0
local function saveSettings()
    saveToken += 1
    local mine = saveToken
    task.delay(0.4, function()
        if mine ~= saveToken then
            return
        end
        local ok, raw = pcall(HttpService.JSONEncode, HttpService, S)
        if not ok then
            return
        end
        pcall(LP.SetAttribute, LP, "AgeHUDLiteSettings", raw)
        if type(writefile) == "function" then
            pcall(writefile, FILE, raw)
        end
    end)
end

local CharacterConfig

local function tryRequire(parts)
    local n = RS
    for _, p in ipairs(parts) do
        n = n and n:FindFirstChild(p)
    end
    if n and n:IsA("ModuleScript") then
        local ok, mod = pcall(require, n)
        if ok and type(mod) == "table" then
            return mod
        end
    end
end

for _, path in ipairs({
    { "CharacterConfig" },
    { "Modules", "CharacterConfig" },
    { "Shared", "CharacterConfig" },
    { "Configs", "CharacterConfig" },
    { "Config", "CharacterConfig" },
    { "Data", "CharacterConfig" },
}) do
    CharacterConfig = tryRequire(path)
    if CharacterConfig then
        break
    end
end

if not CharacterConfig then
    for _, d in ipairs(RS:GetDescendants()) do
        if d:IsA("ModuleScript") and d.Name:lower():find("character") then
            local ok, mod = pcall(require, d)
            if ok and type(mod) == "table" then
                local k = next(mod)
                if k and type(mod[k]) == "table" and mod[k].Age ~= nil then
                    CharacterConfig = mod
                    break
                end
            end
        end
    end
end

local function lookup(name)
    if not CharacterConfig or not name then
        return
    end
    name = tostring(name):gsub("^%s+", ""):gsub("%s+$", "")
    if CharacterConfig[name] then
        return CharacterConfig[name], name
    end
    local lower = name:lower()
    for n, data in pairs(CharacterConfig) do
        if n:lower() == lower then
            return data, n
        end
    end
    for n, data in pairs(CharacterConfig) do
        if n:lower():find(lower, 1, true) or lower:find(n:lower(), 1, true) then
            return data, n
        end
    end
end

local function headingText()
    local g = PG:FindFirstChild("GameGui")
    local h = g and g:FindFirstChild("Heading")
    if not h then
        return
    end
    if h:IsA("TextLabel") or h:IsA("TextButton") or h:IsA("TextBox") then
        return h.Text
    end
    local lab = h:FindFirstChildWhichIsA("TextLabel", true)
        or h:FindFirstChildWhichIsA("TextButton", true)
    return lab and lab.Text
end

local function tween(obj, props, t, style, dir)
    t = t or 0.2
    if t == 0 then
        for k, v in pairs(props) do
            obj[k] = v
        end
        return
    end
    TweenService:Create(
        obj,
        TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    ):Play()
end

local function make(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do
        o[k] = v
    end
    o.Parent = parent
    return o
end

local function openPanel(panel, open, height, instant)
    local d = instant and 0 or 0.3
    if open then
        panel.Visible = true
        tween(panel, { Size = UDim2.fromOffset(WIDTH, height) }, d, Enum.EasingStyle.Quint)
    else
        tween(panel, { Size = UDim2.fromOffset(WIDTH, 0) }, d, Enum.EasingStyle.Quint)
        if instant then
            panel.Visible = false
        else
            task.delay(d + 0.02, function()
                if panel.Size.Y.Offset <= 1 then
                    panel.Visible = false
                end
            end)
        end
    end
end

local function histHeight()
    return HIST_PAD + math.max(#history, 1) * ROW_H
end

local function renderHistory()
    for i = 1, MAX_HIST do
        local e, r = history[i], histRows[i]
        r[1].Visible = e ~= nil
        r[2].Visible = e ~= nil
        if e then
            r[1].Text = e.name
            r[2].Text = tostring(e.age)
        end
    end
    emptyLbl.Visible = #history == 0
    if histOpen then
        openPanel(histPanel, true, histHeight())
    end
end

local function pushHistory(name, age)
    for i, e in ipairs(history) do
        if e.name == name then
            table.remove(history, i)
            break
        end
    end
    table.insert(history, 1, { name = name, age = age })
    while #history > MAX_HIST do
        table.remove(history)
    end
    renderHistory()
end

----------------------------------------------------------------
-- Drag (with edge snapping) + slider dragging
----------------------------------------------------------------
do
    local dragging, moved, start, startPos = false, false, nil, nil

    local function snap(x, y)
        local vp, sz = viewport(), root.AbsoluteSize
        if x < SNAP + MARGIN then
            x = MARGIN
        elseif x + sz.X > vp.X - SNAP - MARGIN then
            x = vp.X - sz.X - MARGIN
        end
        if y < SNAP + TOP_M then
            y = TOP_M
        elseif y + sz.Y > vp.Y - SNAP - MARGIN then
            y = vp.Y - sz.Y - MARGIN
        end
        return x, y
    end

    track(card.InputBegan:Connect(function(input)
        if isPress(input) then
            dragging, moved = true, false
            start = input.Position
            startPos = root.Position
        end
    end))

    track(UserInputService.InputChanged:Connect(function(input)
        if not isMove(input) then
            return
        end
        if activeSlider then
            activeSlider(input.Position.X)
            return
        end
        if not dragging then
            return
        end
        local dx, dy = input.Position.X - start.X, input.Position.Y - start.Y
        if not moved and (dx * dx + dy * dy) > 9 then
            moved = true
        end
        if moved then
            local vp, sz = viewport(), root.AbsoluteSize
            restPos = UDim2.fromOffset(
                math.clamp(startPos.X.Offset + dx, 0, math.max(vp.X - sz.X, 0)),
                math.clamp(startPos.Y.Offset + dy, 0, math.max(vp.Y - sz.Y, 0))
            )
            root.Position = restPos
        end
    end))

    track(UserInputService.InputEnded:Connect(function(input)
        if not isPress(input) then
            return
        end
        if activeSlider then
            activeSlider = nil
            saveSettings()
        end
        if dragging then
            dragging = false
            if moved then
                local x, y = snap(restPos.X.Offset, restPos.Y.Offset)
                restPos = UDim2.fromOffset(x, y)
                tween(root, { Position = restPos }, 0.2, Enum.EasingStyle.Quint)
                S.x, S.y = x, y
                saveSettings()
            end
        end
    end))
end

----------------------------------------------------------------
-- Update logic
----------------------------------------------------------------
local lastName

local function setAnswer(realName, data)
    if not data then
        nameLbl.Text = "Waiting…"
        workLbl.Text = CharacterConfig and "" or "CharacterConfig missing"
        ageLbl.Text = "–"
        setAgeColor("sub")
        return
    end

    nameLbl.Text = realName
    workLbl.Text = data.Work or ""
    ageLbl.Text = tostring(data.Age)
    setAgeColor("accent")

    if realName ~= lastName then
        lastName = realName
        pushHistory(realName, data.Age)
        if shown then
            ageLbl.TextTransparency = 1
            tween(ageLbl, { TextTransparency = 0 }, 0.25)
        end
    end
end

refresh = function()
    if pinned then
        return
    end
    local text = headingText()
    if not text or #text < 2 then
        setAnswer(nil, nil)
        return
    end
    local data, real = lookup(text)
    if data then
        setAnswer(real, data)
    else
        nameLbl.Text = text
        workLbl.Text = "Not in CharacterConfig"
        ageLbl.Text = "?"
        setAgeColor("warn")
    end
end

-- poll + hook Heading when it appears
task.spawn(function()
    while gui.Parent do
        refresh()
        local g = PG:FindFirstChild("GameGui")
        local h = g and g:FindFirstChild("Heading")
        if h and not h:GetAttribute("LiteHooked") then
            h:SetAttribute("LiteHooked", true)
            local function hook(obj)
                if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                    obj:GetPropertyChangedSignal("Text"):Connect(refresh)
                end
            end
            hook(h)
            for _, d in ipairs(h:GetDescendants()) do
                hook(d)
            end
            h.DescendantAdded:Connect(hook)
        end
        task.wait(0.3)
    end
end)
