local TIER_RARITY = {
    [1] = { "Common", Color3.fromRGB(200, 200, 200) },
    [2] = { "Uncommon", Color3.fromRGB(60, 200, 100) },
    [3] = { "Rare", Color3.fromRGB(70, 140, 255) },
    [4] = { "Epic", Color3.fromRGB(180, 70, 255) },
    [5] = { "Legendary", Color3.fromRGB(255, 180, 40) },
}

local function rarityFromTier(tier, name)
    tier = math.clamp(tonumber(tier) or 1, 1, 10)
    local n = string.lower(tostring(name or ""))
    if n:find("mythic", 1, true) or tier >= 6 then
        return "Mythic", Color3.fromRGB(255, 60, 200), tier
    end
    local row = TIER_RARITY[tier] or TIER_RARITY[1]
    return row[1], row[2], tier
end

local function allowed(rarity)
    if rarity == "Common" then return CFG.showCommon end
    if rarity == "Uncommon" then return CFG.showUncommon end
    if rarity == "Rare" then return CFG.showRare end
    if rarity == "Epic" then return CFG.showEpic end
    if rarity == "Legendary" then return CFG.showLegendary end
    if rarity == "Mythic" then return CFG.showMythic end
    return true
end

----------------------------------------------------------------
-- Detect stock / sellable display items
----------------------------------------------------------------
local function looksLikeStock(inst)
    if not inst:IsA("Model") then return false end
    local n = inst.Name
    if n:sub(1, 6) == "Stock_" then return true end
    if inst:GetAttribute("Tier") ~= nil and (
        inst:GetAttribute("DefinitionId") ~= nil
        or inst:GetAttribute("BaseName") ~= nil
        or inst:GetAttribute("Category") ~= nil
    ) then
        return true
    end
    return false
end

local function readItem(inst)
    local tier = inst:GetAttribute("Tier") or 1
    local cat = inst:GetAttribute("Category") or ""
    local name = inst:GetAttribute("DisplayName")
        or inst:GetAttribute("BaseName")
        or inst:GetAttribute("DefinitionId")
        or inst.Name:gsub("^Stock_", ""):gsub("_Dirty$", ""):gsub("_Clean$", ""):gsub("_Display$", "")
    local rarity, color, tnum = rarityFromTier(tier, name)
    return {
        tier = tnum,
        category = tostring(cat),
        name = tostring(name),
        rarity = rarity,
        color = color,
        area = inst:GetAttribute("StenderId") or (inst.Parent and inst.Parent.Parent and inst.Parent.Parent.Name) or "",
    }
end

local function corner(parent, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r)
    c.Parent = parent
end

local function stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = parent
    return s
end

local function label(parent, props)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Font = props.font or FONT
    l.TextSize = props.size or 14
    l.TextColor3 = props.color or THEME.text
    l.TextXAlignment = props.alignX or Enum.TextXAlignment.Left
    l.TextYAlignment = props.alignY or Enum.TextYAlignment.Center
    l.TextTruncate = Enum.TextTruncate.AtEnd
    l.Text = props.text or ""
    l.Size = props.size2 or UDim2.fromScale(1, 1)
    l.Position = props.pos or UDim2.new()
    l.Parent = parent
    return l
end

local alertGui = Instance.new("ScreenGui")

alertGui.DisplayOrder = 120
alertGui.IgnoreGuiInset = true

local alertQueue = {}
local alertBusy = false
local seenKeys = {}
local SOUND_IDS = {
    Mythic = "rbxassetid://4590657391",
    Legendary = "rbxassetid://4590657391",
    Epic = "rbxassetid://6026984224",
    Rare = "rbxassetid://6026984224",
}

local function rarityWantsAlert(rarity)
    if rarity == "Uncommon" then return CFG.alertUncommon end
    if rarity == "Rare" then return CFG.alertRare end
    if rarity == "Epic" then return CFG.alertEpic end
    if rarity == "Legendary" then return CFG.alertLegendary end
    if rarity == "Mythic" then return CFG.alertMythic end
    return false
end

local function playAlertSound(rarity)
    if not CFG.alertSound then return end
    local id = SOUND_IDS[rarity] or SOUND_IDS.Rare
    local s = Instance.new("Sound")
    s.SoundId = id
    s.Volume = rarity == "Mythic" and 1.2 or 0.85
    s.Parent = alertGui
    pcall(function() s:Play() end)
    task.delay(3, function() s:Destroy() end)
end

-- brief screen-edge flash, only for the top two rarities
local function edgeFlash(color)
    local f = Instance.new("Frame")
    f.Size = UDim2.fromScale(1, 1)
    f.BackgroundTransparency = 1
    f.BorderSizePixel = 0
    f.Parent = alertGui
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = 10
    s.Transparency = 0.5
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = f
    local tw = TweenService:Create(s, TweenInfo.new(0.7, Enum.EasingStyle.Quad), { Transparency = 1 })
    tw:Play()
    tw.Completed:Connect(function() f:Destroy() end)
end

local function runNextAlert()
    if alertBusy or #alertQueue == 0 then return end
    alertBusy = true
    local info = table.remove(alertQueue, 1)
    local color = RARITY_COLORS[info.rarity] or THEME.on
    local big = info.rarity == "Mythic" or info.rarity == "Legendary"
    local W, H = big and 420 or 340, big and 76 or 62

    local holder = Instance.new("Frame")
    holder.AnchorPoint = Vector2.new(0.5, 0)
    holder.Position = UDim2.new(0.5, 0, 0, -H - 10)
    holder.Size = UDim2.fromOffset(W, H)
    holder.BackgroundColor3 = THEME.bg
    holder.BorderSizePixel = 0
    holder.Parent = alertGui
    corner(holder, 6)
    stroke(holder, big and color or THEME.stroke, big and 2 or 1, 0)

    label(holder, {
        text = info.rarity, font = FONT_BOLD, size = big and 16 or 14, color = color,
        size2 = UDim2.new(0.5, -14, 0, 20), pos = UDim2.fromOffset(14, big and 10 or 8),
    })
    label(holder, {
        text = string.format("T%d  ·  %s", info.tier, info.category ~= "" and info.category or "Store"),
        font = FONT, size = 13, color = THEME.dim, alignX = Enum.TextXAlignment.Right,
        size2 = UDim2.new(0.5, -14, 0, 20), pos = UDim2.new(0.5, 0, 0, big and 10 or 8),
    })
    label(holder, {
        text = info.name, font = FONT_BOLD, size = big and 24 or 19, color = THEME.text,
        size2 = UDim2.new(1, -28, 0, big and 32 or 26), pos = UDim2.fromOffset(14, big and 34 or 29),
    })

    playAlertSound(info.rarity)
    if big then edgeFlash(color) end

    local duration = big and 4.5 or 3.2
    TweenService:Create(holder, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, 0, 0, 20),
    }):Play()

    task.delay(duration, function()
        local tw = TweenService:Create(holder, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Position = UDim2.new(0.5, 0, 0, -H - 10),
        })
        tw:Play()
        tw.Completed:Wait()
        holder:Destroy()
        alertBusy = false
        runNextAlert()
    end)
end

local function notifyNew(inst, info)
    if not CFG.enabled then return end
    if info.tier < CFG.minTier then return end
    if not allowed(info.rarity) then return end
    if not rarityWantsAlert(info.rarity) then return end
    local key = tostring(inst)
    if seenKeys[key] then return end
    seenKeys[key] = true
    table.insert(alertQueue, info)
    runNextAlert()
end

----------------------------------------------------------------
-- ESP tracking
----------------------------------------------------------------
local tracked = {}

local function clearOne(inst)
    local t = tracked[inst]
    if not t then return end
    if t.hl then pcall(function() t.hl:Destroy() end) end
    if t.bill then pcall(function() t.bill:Destroy() end) end
    tracked[inst] = nil
end

local function clearAll()
    for inst in pairs(tracked) do clearOne(inst) end
end

local function shouldShow(info)
    if not CFG.enabled then return false end
    if info.tier < CFG.minTier then return false end
    return allowed(info.rarity)
end

local function makeNameText(info)
    return CFG.showName and info.name or ""
end

local function makeMetaText(info)
    local parts = {}
    if CFG.showTier then table.insert(parts, string.format("%s T%d", info.rarity, info.tier)) end
    if CFG.showCategory and info.category ~= "" then table.insert(parts, info.category) end
    return table.concat(parts, " · ")
end

local function ensure(inst)
    if not looksLikeStock(inst) then return end
    local info = readItem(inst)
    if not shouldShow(info) then
        clearOne(inst)
        return
    end

    local t = tracked[inst]
    if t then
        if t.hl then
            t.hl.FillColor = info.color
            t.hl.OutlineColor = info.color
            t.hl.Enabled = true
        end
        if t.nameLabel then t.nameLabel.Text = makeNameText(info) end
        if t.metaLabel then
            t.metaLabel.Text = makeMetaText(info)
            t.metaLabel.TextColor3 = info.color
        end
        t.info = info
        return
    end

    local hl = Instance.new("Highlight")
    hl.Name = "ThriftRarityESP"
    hl.Adornee = inst
    hl.FillColor = info.color
    hl.OutlineColor = info.color
    hl.FillTransparency = CFG.fillTransparency
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = inst

    local bill = Instance.new("BillboardGui")
    bill.Name = "ThriftRarityTag"
    bill.Size = UDim2.fromOffset(180, 34)
    bill.StudsOffset = Vector3.new(0, 2.4, 0)
    bill.AlwaysOnTop = true
    bill.MaxDistance = CFG.maxDistance
    local anchor = inst:FindFirstChild("Root", true)
        or inst.PrimaryPart
        or inst:FindFirstChildWhichIsA("BasePart", true)
        or inst
    bill.Adornee = anchor
    bill.Parent = inst

    -- plain outlined text, no box
    local nameLabel = label(bill, {
        text = makeNameText(info), font = FONT_BOLD, size = 14, color = THEME.text,
        alignX = Enum.TextXAlignment.Center,
        size2 = UDim2.new(1, 0, 0, 18), pos = UDim2.fromOffset(0, 0),
    })
    nameLabel.TextStrokeTransparency = 0.2
    nameLabel.TextStrokeColor3 = Color3.new(0, 0, 0)

    local metaLabel = label(bill, {
        text = makeMetaText(info), font = FONT_MED, size = 12, color = info.color,
        alignX = Enum.TextXAlignment.Center,
        size2 = UDim2.new(1, 0, 0, 14), pos = UDim2.fromOffset(0, 18),
    })
    metaLabel.TextStrokeTransparency = 0.2
    metaLabel.TextStrokeColor3 = Color3.new(0, 0, 0)

    tracked[inst] = { hl = hl, bill = bill, nameLabel = nameLabel, metaLabel = metaLabel, info = info }
    notifyNew(inst, info)
end

local function scan()
    local root = getThriftRoot()
    if not root then return end

    local keep = {}
    for _, d in ipairs(root:GetDescendants()) do
        if looksLikeStock(d) then
            ensure(d)
            keep[d] = true
        end
    end
    for inst in pairs(tracked) do
        if not keep[inst] or not inst.Parent then
            clearOne(inst)
        end
    end
end

task.defer(function()
    local root = getThriftRoot()
    if not root then return end
    root.DescendantAdded:Connect(function(d)
        if looksLikeStock(d) then
            task.defer(ensure, d)
        end
    end)
    root.DescendantRemoving:Connect(function(d)
        if tracked[d] then clearOne(d) end
    end)
end)

task.spawn(function()
    while true do
        if CFG.enabled then scan() end
        task.wait(1.2)
    end
end)

RunService.RenderStepped:Connect(function()
    if not CFG.enabled then return end
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local origin = hrp.Position
    for inst, t in pairs(tracked) do
        local pos
        if inst.PrimaryPart then
            pos = inst.PrimaryPart.Position
        else
            local p = inst:FindFirstChild("Root", true) or inst:FindFirstChildWhichIsA("BasePart", true)
            pos = p and p.Position
        end
        if pos and t.hl then
            local on = (pos - origin).Magnitude <= CFG.maxDistance
            t.hl.Enabled = on
            if t.bill then t.bill.Enabled = on end
        end
    end
end)

        countLabels[rarity] = label(row, {

            size2 = UDim2.new(0, 30, 1, 0), pos = UDim2.new(1, -52, 0, 0),

    box.Size = UDim2.fromOffset(16, 16)

    box.Position = UDim2.new(1, -2, 0.5, 0)

    box.Parent = row
    corner(box, 3)
    local boxStroke = stroke(box, THEME.stroke, 1, 0)
    local mark = label(box, {
        text = "✓", font = FONT_BOLD, size = 12, color = Color3.new(1, 1, 1),
        alignX = Enum.TextXAlignment.Center,
    })

    local function render()
        local on = CFG[key]
        box.BackgroundColor3 = on and (rarity and RARITY_COLORS[rarity] or THEME.on) or THEME.row
        boxStroke.Transparency = on and 1 or 0
        mark.Visible = on
    end
    render()

    row.MouseButton1Click:Connect(function()
        CFG[key] = not CFG[key]
        render()
        if key == "enabled" and not CFG[key] then clearAll() end
        scan()
    end)
end

-- drag
do
    local dragging, dragStart, startPos
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end
