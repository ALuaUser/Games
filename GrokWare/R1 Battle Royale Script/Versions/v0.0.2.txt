local function char()
    return LocalPlayer.Character
end

local function hrp(c)
    c = c or char()
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function hum(c)
    c = c or char()
    return c and c:FindFirstChildOfClass("Humanoid")
end

local function notify(title, content, dur)
    Library:Notify({ Title = title, Content = content, Duration = dur or 4 })
end

local function isAlive(model)
    local h = model and model:FindFirstChildOfClass("Humanoid")
    return h and h.Health > 0
end

local function isTeammate(plr)
    if not plr or plr == LocalPlayer then return false end
    local myTeam = LocalPlayer:GetAttribute("TPS_Team") or LocalPlayer:GetAttribute("TeamId")
    local theirTeam = plr:GetAttribute("TPS_Team") or plr:GetAttribute("TeamId")
    if myTeam and theirTeam and myTeam == theirTeam then return true end
    local mate = LocalPlayer:GetAttribute("TPS_MateName")
    if mate and (plr.Name == mate or plr.DisplayName == mate) then return true end
    return false
end

local ESPFolder = Instance.new("Folder")
ESPFolder.Name = "R1_EvolESP"
ESPFolder.Parent = game:GetService("CoreGui")

local espObjects = {}

local function clearESP(model)
    local obj = espObjects[model]
    if not obj then return end
    for _, v in pairs(obj) do
        pcall(function() v:Destroy() end)
    end
    espObjects[model] = nil
end

local function clearAllESP()
    for model in pairs(espObjects) do
        clearESP(model)
    end
end

local function makeESP(model, color, labelText)
    if espObjects[model] then clearESP(model) end
    if not model or not model.Parent then return end

    local root = model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart")
    if not root then return end

    local data = {}

    if CFG.ShowChams then
        local hl = Instance.new("Highlight")
        hl.Name = "EvolChams"
        hl.Adornee = model
        hl.FillColor = color
        hl.OutlineColor = Color3.new(1, 1, 1)
        hl.FillTransparency = 0.55
        hl.OutlineTransparency = 0.2
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Parent = ESPFolder
        data.hl = hl
    end

    local bb = Instance.new("BillboardGui")
    bb.Name = "EvolTag"
    bb.Adornee = root
    bb.Size = UDim2.fromOffset(160, 40)
    bb.StudsOffset = Vector3.new(0, 3.2, 0)
    bb.AlwaysOnTop = true
    bb.Parent = ESPFolder

    local text = Instance.new("TextLabel")
    text.Size = UDim2.fromScale(1, 1)
    text.BackgroundTransparency = 1
    text.Text = labelText or model.Name
    text.TextColor3 = color
    text.TextStrokeTransparency = 0.4
    text.Font = Enum.Font.GothamBold
    text.TextSize = 13
    text.Parent = bb

    data.bb = bb
    data.label = text
    data.color = color
    data.root = root
    espObjects[model] = data
end

local LOOT_KEYWORDS = {
    "Armor", "Helmet", "Weapon", "Gun", "Rifle", "Pistol", "Shotgun",
    "SMG", "Sniper", "Med", "Bandage", "Syringe", "Ammo", "Crate",
    "Supply", "Backpack", "Vest", "Pickup", "Loot",
}

local function isLootModel(m)
    if not m or not m:IsA("Model") then return false end
    local n = m.Name
    for _, k in ipairs(LOOT_KEYWORDS) do
        if string.find(n, k, 1, true) then return true end
    end
    if m:FindFirstChildWhichIsA("ProximityPrompt", true) and not m:FindFirstChildOfClass("Humanoid") then
        return true
    end
    return false
end

local function scanESP()
    if not (CFG.PlayerESP or CFG.BotESP or CFG.LootESP) then
        -- don't clear gas visuals here
        for model in pairs(espObjects) do
            if model ~= "GAS_CURRENT" and model ~= "GAS_NEXT" then
                clearESP(model)
            end
        end
        return
    end

    local myRoot = hrp()
    local seen = {}

    if CFG.PlayerESP then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character and isAlive(plr.Character) then
                local model = plr.Character
                seen[model] = true
                local color = isTeammate(plr) and CFG.TeamColor or CFG.PlayerColor
                local dist = ""
                if CFG.ShowDistance and myRoot then
                    local r = hrp(model)
                    if r then
                        dist = string.format(" [%dm]", math.floor((myRoot.Position - r.Position).Magnitude))
                    end
                end
                local name = CFG.ShowNames
                    and (plr.DisplayName ~= plr.Name and (plr.DisplayName .. " (@" .. plr.Name .. ")") or plr.Name)
                    or "Player"
                makeESP(model, color, name .. dist)
            end
        end
    end

    if CFG.BotESP then
        local bots = Workspace:FindFirstChild("TPS_Bots")
        if bots then
            for _, model in ipairs(bots:GetChildren()) do
                if model:IsA("Model") and isAlive(model) then
                    seen[model] = true
                    local dist = ""
                    if CFG.ShowDistance and myRoot then
                        local r = hrp(model) or model.PrimaryPart
                        if r then
                            dist = string.format(" [%dm]", math.floor((myRoot.Position - r.Position).Magnitude))
                        end
                    end
                    makeESP(model, CFG.BotColor, (CFG.ShowNames and ("[BOT] " .. model.Name) or "[BOT]") .. dist)
                end
            end
        end
    end

    if CFG.LootESP then
        local function scanFolder(folder, depth)
            if depth > 4 or not folder then return end
            for _, child in ipairs(folder:GetChildren()) do
                if isLootModel(child) then
                    seen[child] = true
                    local root = child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart")
                    local dist = ""
                    if CFG.ShowDistance and myRoot and root then
                        dist = string.format(" [%dm]", math.floor((myRoot.Position - root.Position).Magnitude))
                    end
                    makeESP(child, CFG.LootColor, (CFG.ShowNames and child.Name or "Loot") .. dist)
                elseif child:IsA("Folder") or child:IsA("Model") then
                    scanFolder(child, depth + 1)
                end
            end
        end
        scanFolder(Workspace:FindFirstChild("Map"), 0)
        scanFolder(Workspace:FindFirstChild("TPS_Range"), 0)
        scanFolder(Workspace:FindFirstChild("Pickups"), 0)
        for _, child in ipairs(Workspace:GetChildren()) do
            if child.Name:find("Loot") or child.Name:find("Pickup") or child.Name:find("Supply") then
                scanFolder(child, 0)
            end
        end
    end

    for model in pairs(espObjects) do
        if model ~= "GAS_CURRENT" and model ~= "GAS_NEXT" then
            if not seen[model] or not model.Parent then
                clearESP(model)
            end
        end
    end
end

RunService.RenderStepped:Connect(function()
    if not (CFG.PlayerESP or CFG.BotESP or CFG.LootESP) then return end
    local myRoot = hrp()
    if not myRoot then return end
    for model, obj in pairs(espObjects) do
        if type(model) == "string" then continue end
        if obj.root and obj.root.Parent then
            local d = (myRoot.Position - obj.root.Position).Magnitude
            local show = d <= CFG.MaxDistance
            if obj.hl then obj.hl.Enabled = show and CFG.ShowChams end
            if obj.bb then obj.bb.Enabled = show end
            if show and obj.label and CFG.ShowDistance then
                local base = obj.label.Text:gsub("%s*%[%d+m%]", "")
                obj.label.Text = base .. string.format(" [%dm]", math.floor(d))
            end
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(1.25)
        pcall(scanESP)
    end
end)

--[[
    Tries multiple sources common in R1 / TPS:
    - Workspace or RS: MatchState / Gas / Zone attributes
    - NumberValues / Vector3Values named *Gas* *Zone* *Safe*
    - Falls back to any Part/Model named Gas, Zone, SafeZone, Storm
]]

local gasParts = {
    current = nil, -- ring or cylinder visual
    next = nil,
    label = nil,
}

local function destroyGasVisuals()
    for _, v in pairs(gasParts) do
        pcall(function()
            if typeof(v) == "Instance" then v:Destroy() end
        end)
    end
    gasParts = { current = nil, next = nil, label = nil }
end

local function findMatchState()
    local candidates = {
        Workspace:FindFirstChild("MatchState"),
        ReplicatedStorage:FindFirstChild("MatchState"),
        Workspace:FindFirstChild("TPS_MatchState"),
        ReplicatedStorage:FindFirstChild("TPS") and ReplicatedStorage.TPS:FindFirstChild("MatchState"),
        Workspace:FindFirstChild("Gas"),
        Workspace:FindFirstChild("Zone"),
    }
    for _, c in ipairs(candidates) do
        if c then return c end
    end
    -- attribute host on Workspace
    return Workspace
end

local function readGasData()
    local ms = findMatchState()
    local center, radius, nextCenter, nextRadius

    local function attr(inst, names)
        for _, n in ipairs(names) do
            local v = inst:GetAttribute(n)
            if v ~= nil then return v end
        end
    end

    local function val(inst, names)
        for _, n in ipairs(names) do
            local o = inst:FindFirstChild(n, true)
            if o and (o:IsA("Vector3Value") or o:IsA("NumberValue") or o:IsA("NumberValue")) then
                return o.Value
            end
        end
    end

    if ms then
        center = attr(ms, { "GasCenter", "ZoneCenter", "SafeCenter", "StormCenter", "CircleCenter", "R1_GasCenter" })
            or val(ms, { "GasCenter", "ZoneCenter", "SafeCenter" })
        radius = attr(ms, { "GasRadius", "ZoneRadius", "SafeRadius", "StormRadius", "CircleRadius", "R1_GasRadius" })
            or val(ms, { "GasRadius", "ZoneRadius", "SafeRadius" })
        nextCenter = attr(ms, { "NextGasCenter", "NextZoneCenter", "NextSafeCenter", "R1_NextGasCenter" })
            or val(ms, { "NextGasCenter", "NextZoneCenter" })
        nextRadius = attr(ms, { "NextGasRadius", "NextZoneRadius", "NextSafeRadius", "R1_NextGasRadius" })
            or val(ms, { "NextGasRadius", "NextZoneRadius" })
    end

    -- Workspace attributes
    if not center then
        center = Workspace:GetAttribute("GasCenter") or Workspace:GetAttribute("ZoneCenter")
        radius = Workspace:GetAttribute("GasRadius") or Workspace:GetAttribute("ZoneRadius")
    end

    -- Physical zone parts
    if not center or not radius then
        for _, name in ipairs({ "SafeZone", "GasZone", "Storm", "Gas", "Zone", "Circle" }) do
            local p = Workspace:FindFirstChild(name, true)
            if p and p:IsA("BasePart") then
                center = center or p.Position
                radius = radius or (math.max(p.Size.X, p.Size.Z) / 2)
                break
            elseif p and p:IsA("Model") then
                local pp = p.PrimaryPart or p:FindFirstChildWhichIsA("BasePart")
                if pp then
                    center = center or pp.Position
                    radius = radius or (math.max(pp.Size.X, pp.Size.Z) / 2)
                    break
                end
            end
        end
    end

    if typeof(center) == "Vector3" and type(radius) == "number" and radius > 0 then
        return center, radius, nextCenter, nextRadius
    end
    return nil
end

local function makeRing(radius, color, height)
    height = height or 4
    local part = Instance.new("Part")
    part.Name = "EvolGasRing"
    part.Anchored = true
    part.CanCollide = false
    part.CanQuery = false
    part.CanTouch = false
    part.Material = Enum.Material.ForceField
    part.Color = color
    part.Transparency = 0.55
    part.Size = Vector3.new(radius * 2, height, radius * 2)
    part.Shape = Enum.PartType.Cylinder
    -- Cylinder is Y-axis by default → rotate to lie flat
    part.CFrame = CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, math.rad(90))
    part.Parent = ESPFolder
    return part
end

local function updateGasESP()
    if not CFG.GasESP then
        destroyGasVisuals()
        return
    end

    local center, radius, nextCenter, nextRadius = readGasData()
    if not center or not radius then
        destroyGasVisuals()
        return
    end

    -- Current zone ring
    if not gasParts.current or not gasParts.current.Parent then
        gasParts.current = makeRing(radius, CFG.GasColor, 6)
    end
    gasParts.current.Size = Vector3.new(radius * 2, 6, radius * 2)
    gasParts.current.CFrame = CFrame.new(center) * CFrame.Angles(0, 0, math.rad(90))
    gasParts.current.Color = CFG.GasColor

    -- Next zone (if present)
    if typeof(nextCenter) == "Vector3" and type(nextRadius) == "number" and nextRadius > 0 then
        if not gasParts.next or not gasParts.next.Parent then
            gasParts.next = makeRing(nextRadius, CFG.GasNextColor, 3)
        end
        gasParts.next.Size = Vector3.new(nextRadius * 2, 3, nextRadius * 2)
        gasParts.next.CFrame = CFrame.new(nextCenter) * CFrame.Angles(0, 0, math.rad(90))
        gasParts.next.Color = CFG.GasNextColor
        gasParts.next.Transparency = 0.7
    elseif gasParts.next then
        pcall(function() gasParts.next:Destroy() end)
        gasParts.next = nil
    end

    -- Billboard at center
    if not gasParts.label or not gasParts.label.Parent then
        local bb = Instance.new("BillboardGui")
        bb.Size = UDim2.fromOffset(200, 50)
        bb.AlwaysOnTop = true
        bb.StudsOffset = Vector3.new(0, 12, 0)
        bb.Parent = ESPFolder
        local t = Instance.new("TextLabel")
        t.Size = UDim2.fromScale(1, 1)
        t.BackgroundTransparency = 1
        t.TextColor3 = CFG.GasColor
        t.TextStrokeTransparency = 0.3
        t.Font = Enum.Font.GothamBold
        t.TextSize = 14
        t.Parent = bb
        gasParts.label = bb
        gasParts.labelText = t
    end
    gasParts.label.Adornee = gasParts.current
    local myRoot = hrp()
    local dist = myRoot and math.floor((myRoot.Position - center).Magnitude) or 0
    local inside = myRoot and (Vector3.new(myRoot.Position.X, 0, myRoot.Position.Z) - Vector3.new(center.X, 0, center.Z)).Magnitude <= radius
    gasParts.labelText.Text = string.format(
        "GAS  r=%.0f  %dm %s",
        radius,
        dist,
        inside and "INSIDE" or "OUTSIDE"
    )
    gasParts.labelText.TextColor3 = inside and Color3.fromRGB(80, 255, 120) or CFG.GasColor
end

task.spawn(function()
    while true do
        task.wait(0.5)
        pcall(updateGasESP)
    end
end)

local function getVehicle()
    local c = char()
    if not c then return nil end
    local h = hum(c)
    if not h or not h.SeatPart then return nil end

    local seat = h.SeatPart
    -- climb to vehicle model
    local model = seat
    for _ = 1, 6 do
        if model:IsA("Model") and (model:FindFirstChildWhichIsA("VehicleSeat") or model:FindFirstChild("DriveSeat") or model.Name:lower():find("car") or model.Name:lower():find("vehicle")) then
            break
        end
        if not model.Parent or model.Parent == Workspace then break end
        model = model.Parent
    end

    local root = model:FindFirstChild("Chassis")
        or model:FindFirstChild("Body")
        or model.PrimaryPart
        or model:FindFirstChild("HumanoidRootPart")
        or seat
    if root and root:IsA("BasePart") then
        return model, root, seat
    end
    return nil
end

local flyKeys = { W = false, A = false, S = false, D = false, Space = false, LeftShift = false }

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    local k = input.KeyCode
    if k == Enum.KeyCode.W then flyKeys.W = true
    elseif k == Enum.KeyCode.A then flyKeys.A = true
    elseif k == Enum.KeyCode.S then flyKeys.S = true
    elseif k == Enum.KeyCode.D then flyKeys.D = true
    elseif k == Enum.KeyCode.Space then flyKeys.Space = true
    elseif k == Enum.KeyCode.LeftShift then flyKeys.LeftShift = true
    end
end)

UserInputService.InputEnded:Connect(function(input)
    local k = input.KeyCode
    if k == Enum.KeyCode.W then flyKeys.W = false
    elseif k == Enum.KeyCode.A then flyKeys.A = false
    elseif k == Enum.KeyCode.S then flyKeys.S = false
    elseif k == Enum.KeyCode.D then flyKeys.D = false
    elseif k == Enum.KeyCode.Space then flyKeys.Space = false
    elseif k == Enum.KeyCode.LeftShift then flyKeys.LeftShift = false
    end
end)

RunService.Heartbeat:Connect(function(dt)
    if not (CFG.VehSpeed or CFG.VehFly) then return end
    local model, root = getVehicle()
    if not root then return end

    if CFG.VehFly then
        local cam = Camera.CFrame
        local dir = Vector3.zero
        if flyKeys.W then dir += cam.LookVector end
        if flyKeys.S then dir -= cam.LookVector end
        if flyKeys.A then dir -= cam.RightVector end
        if flyKeys.D then dir += cam.RightVector end
        if flyKeys.Space then dir += Vector3.yAxis end
        if flyKeys.LeftShift then dir -= Vector3.yAxis end

        if dir.Magnitude > 0 then
            dir = dir.Unit * CFG.VehFlySpeed
        end
        -- smooth velocity write
        pcall(function()
            root.AssemblyLinearVelocity = dir
            -- keep upright-ish
            local look = Vector3.new(cam.LookVector.X, 0, cam.LookVector.Z)
            if look.Magnitude > 0.1 then
                root.CFrame = CFrame.new(root.Position, root.Position + look)
            end
        end)
    elseif CFG.VehSpeed then
        local vel = root.AssemblyLinearVelocity
        local horizontal = Vector3.new(vel.X, 0, vel.Z)
        if horizontal.Magnitude > 5 then
            local boost = horizontal.Unit * (horizontal.Magnitude * CFG.VehSpeedMult)
            pcall(function()
                root.AssemblyLinearVelocity = Vector3.new(boost.X, vel.Y, boost.Z)
            end)
        end
    end
end)

local noclipConn
local function setNoclip(on)
    if noclipConn then
        noclipConn:Disconnect()
        noclipConn = nil
    end
    if not on then return end
    noclipConn = RunService.Stepped:Connect(function()
        local c = char()
        if not c then return end
        for _, part in ipairs(c:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)
end

RunService.Heartbeat:Connect(function()
    local h = hum()
    if not h then return end
    if CFG.SpeedEnabled then h.WalkSpeed = CFG.Speed end
    if CFG.JumpEnabled then
        pcall(function()
            h.JumpPower = CFG.JumpPower
            h.UseJumpPower = true
        end)
    end
end)

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    if CFG.Noclip then setNoclip(true) end
end)

UserInputService.JumpRequest:Connect(function()
    if CFG.InfiniteJump then
        local h = hum()
        if h then h:ChangeState(Enum.HumanoidStateType.Jumping) end
    end
end)

local oldLighting = {}
local function applyFullbright(on)
    if on then
        if not oldLighting.Brightness then
            oldLighting.Brightness = Lighting.Brightness
            oldLighting.ClockTime = Lighting.ClockTime
            oldLighting.FogEnd = Lighting.FogEnd
            oldLighting.GlobalShadows = Lighting.GlobalShadows
            oldLighting.Ambient = Lighting.Ambient
            oldLighting.OutdoorAmbient = Lighting.OutdoorAmbient
        end
        Lighting.Brightness = 2
        Lighting.ClockTime = 14
        Lighting.FogEnd = 1e6
        Lighting.GlobalShadows = false
        Lighting.Ambient = Color3.new(1, 1, 1)
        Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
    else
        if oldLighting.Brightness then
            Lighting.Brightness = oldLighting.Brightness
            Lighting.ClockTime = oldLighting.ClockTime
            Lighting.FogEnd = oldLighting.FogEnd
            Lighting.GlobalShadows = oldLighting.GlobalShadows
            Lighting.Ambient = oldLighting.Ambient
            Lighting.OutdoorAmbient = oldLighting.OutdoorAmbient
            oldLighting = {}
        end
    end
end

local function redeemCode(code)
    local rf = ReplicatedStorage:FindFirstChild("R1Codes")
    if not rf then
        notify("Codes", "R1Codes remote not found (use in lobby)", 4)
        return
    end
    local ok, res = pcall(function()
        return rf:InvokeServer(code)
    end)
    notify("Codes", ok and tostring(res or ("Sent: " .. code)) or ("Failed: " .. tostring(res)), 5)
end

    Callback = function()
        local ok, servers = pcall(function()
            return HttpService:JSONDecode(game:HttpGet(
                "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
            ))
        end)
        if not ok or not servers or not servers.data then
            notify("Server Hop", "Failed to fetch servers", 4)
            return
        end
        for _, s in ipairs(servers.data) do
            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                return
            end
        end
        notify("Server Hop", "No open server found", 4)
    end,
})
