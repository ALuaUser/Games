local function httpGet(url)
    local req = (syn and syn.request)
        or (http and http.request)
        or http_request
        or request
        or (fluxus and fluxus.request)

    if req then
        local ok, res = pcall(req, {
            Url = url,
            Method = "GET",
        })
        if ok and res then
            local body = res.Body or res.body
            local code = res.StatusCode or res.status_code or res.Status or 200
            if body and (code == 200 or code == 0) then
                return body
            end
            return nil, "HTTP " .. tostring(code)
        end
        return nil, tostring(res)
    end

    local ok, body = pcall(function()
        return game:HttpGet(url)
    end)
    if ok and body and body ~= "" then
        return body
    end
    return nil, tostring(body)
end

local function urlEncode(s)
    local out = tostring(s):gsub("([^%w%-%.%_%~])", function(c)
        return string.format("%%%02X", string.byte(c))
    end)
    return out
end

local function apiContents(path)
    local parts = {}
    for seg in string.gmatch(path, "[^/]+") do
        parts[#parts + 1] = urlEncode(seg)
    end
    local url = ("https://api.github.com/repos/%s/contents/%s?ref=%s"):format(
        REPO,
        table.concat(parts, "/"),
        BRANCH
    )

    local body, err = httpGet(url)
    if not body then
        return nil, err
    end
    local ok, data = pcall(function()
        return HttpService:JSONDecode(body)
    end)
    if not ok or type(data) ~= "table" then
        return nil, "bad JSON"
    end
    return data
end

local function listVersions(folder)
    local data, err = apiContents(ROOT .. "/" .. folder .. "/Versions")
    if not data then
        return nil, err
    end
    local files = {}
    for _, item in ipairs(data) do
        if item.type == "file" and item.name then
            local n = item.name
            if n:match("%.lua$") or n:match("%.txt$") or n:match("%.luau$") then
                local fallback = ("https://raw.githubusercontent.com/%s/%s/%s"):format(
                    REPO,
                    BRANCH,
                    (item.path:gsub(" ", "%%20"))
                )
                files[#files + 1] = {
                    name = n,
                    path = item.path,
                    download = item.download_url or fallback,
                    size = item.size or 0,
                }
            end
        end
    end
    table.sort(files, function(a, b)
        return a.name > b.name -- newest-looking first (v0.0.4 > v0.0.1)
    end)
    return files
end

local function loadVersion(file)
    local url = file.download
    if not url then
        return false, "no download url"
    end
    url = url:gsub(" ", "%%20")
    local src, err = httpGet(url)
    if not src or src == "" then
        return false, err or "empty response"
    end
    local fn, cerr = loadstring(src)
    if not fn then
        return false, "compile: " .. tostring(cerr)
    end
    local ok, res = pcall(fn)
    if not ok then
        return false, "runtime: " .. tostring(res)
    end
    return true, res
end

-- drag (header only)
do
    local dragging, start, startPos
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            start = input.Position
            startPos = card.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then
            return
        end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - start
            card.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end

local loading = false

local function showUnsupported()
    sub.Text = "Game not supported"
    setStatus("PlaceId is not mapped", "err")
    clearList()
    addRow("Copy PlaceId", nil, function()
        if setclipboard then
            setclipboard(tostring(placeId))
            setStatus("PlaceId copied", "ok")
        end
    end)
    addRow("Copy repo link", nil, function()
        if setclipboard then
            setclipboard("https://github.com/ALuaUser/Games/tree/main/GrokWare")
            setStatus("Repo link copied", "ok")
        end
    end)
end

local function showVersions()
    sub.Text = entry.name
    setStatus("Fetching versions", "idle")
    clearList()

    task.spawn(function()
        local files, err = listVersions(entry.folder)
        if not files then
            setStatus("Failed: " .. tostring(err), "err")
            addRow("Retry", nil, showVersions)
            return
        end
        if #files == 0 then
            setStatus("No scripts in Versions/", "warn")
            return
        end
        setStatus(("%d version%s available"):format(#files, #files == 1 and "" or "s"), "ok")

        for _, file in ipairs(files) do
            local meta = file.size > 0 and string.format("%.1f KB", file.size / 1024) or nil
            addRow(file.name, meta, function()
                if loading then
                    return
                end
                loading = true
                setStatus("Loading " .. file.name, "idle")
                task.spawn(function()
                    local ok, res = loadVersion(file)
                    loading = false
                    if ok then
                        setStatus("Loaded " .. file.name, "ok")
                        task.delay(0.8, function()
                            if gui.Parent then
                                gui:Destroy()
                            end
                        end)
                    else
                        setStatus(tostring(res), "err")
                    end
                end)
            end)
        end
    end)
end
